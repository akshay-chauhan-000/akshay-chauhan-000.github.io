{{- /* Generic mount point for a self-contained browser simulation. Usage: {{< sim "particle-life" >}} {{< sim name="particle-life" height="600" >}} Expects the simulation to live at: static/sims//.css static/sims//.js and to bootstrap itself by querying [data-sim=""]. */ -}} {{- $name := or (.Get "name") (.Get 0) -}} {{- if not $name -}} {{- errorf "sim shortcode: missing simulation name at %s (usage: {{< sim \"particle-life\" >}})" .Position -}} {{- end -}} {{- $css := printf "sims/%s/%s.css" $name $name -}} {{- $js := printf "sims/%s/%s.js" $name $name -}}
